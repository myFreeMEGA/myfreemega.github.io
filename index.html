Ejecución en Servidor Local (Node.js) y MEGA-CMD

Alternativa sin límites de memoria del navegador para carpetas de varios Gigabytes

1. Servidor Express + megajs (API Local) 
// Instalar dependencias: npm install express cors megajs
import express from 'express';
import cors from 'cors';
import { File } from 'megajs';

const app = express();
app.use(cors());
app.use(express.json());

app.post('/api/inspect', async (req, res) => {
  try {
    const { url } = req.body;
    const folder = File.fromURL(url);
    await folder.loadAttributes();

    const serializeNode = (node) => ({
      name: node.name,
      size: node.size || 0,
      directory: !!node.directory,
      timestamp: node.timestamp,
      children: node.children ? node.children.map(serializeNode) : undefined
    });

    res.json(serializeNode(folder));
  } catch (err) {
    res.status(400).json({ error: err.message });
  }
});

app.listen(3000, () => console.log('Servidor MEGA corriendo en http://localhost:3000'));
